The live probe command, run from the repository root (the temporary JPEG is generated locally):

```sh
node --conditions=source --input-type=module <<'JS'
import { readFileSync, writeFileSync } from 'node:fs';
import { createMetaClient } from './services/gateway/src/models/meta-client.ts';
import { parseScreen } from './services/gateway/src/parse/spark-parse.ts';
process.loadEnvFile('.env');
const key = process.env.META_API_KEY;
const result = { syntheticOnly: true, requests: 0, retries: 0, timeoutMs: 5000, completed: false };
if (!key) result.error = 'credential_missing';
else {
 const start = performance.now();
 try {
  result.requests++;
  const parsed = await parseScreen(createMetaClient(key).withOptions({ logLevel: 'off' }),
   {bytes: readFileSync('/tmp/fingerpost-day2-synthetic-probe.jpg'), mime: 'image/jpeg'}, {locale: 'en', goal: 'Start'});
  result.completed = true; result.elements = parsed.layout.elements.length; result.latencyMs = parsed.latencyMs;
 } catch (error) {
  const known = ['APIConnectionTimeoutError','APIConnectionError','AuthenticationError','PermissionDeniedError','RateLimitError','BadRequestError','InternalServerError'];
  result.error = known.includes(error?.constructor?.name) ? error.constructor.name : 'provider_or_response';
  if (Number.isInteger(error?.status)) result.httpStatus = error.status;
 } finally { result.elapsedMs = Math.round(performance.now()-start); }
}
writeFileSync('/tmp/fingerpost-day2-live-probe-result.json',JSON.stringify(result),{mode:0o600});
console.log(JSON.stringify(result));
```
